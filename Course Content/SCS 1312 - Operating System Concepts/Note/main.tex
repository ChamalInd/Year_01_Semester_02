\section{Introduction}

\subsection{Execution Model}
\begin{itemize}
	\item \textbf{Assumption:} a machine with a single-core CPU, one program counter (\reg{PC}) and one instruction register (\reg{IR}).
	\item The CPU repeatedly performs the \emph{fetch--execute} cycle: \reg{PC} holds the address of the next instruction, that instruction is fetched into \reg{IR}, and then it is executed.
	\item When an application is running, the CPU executes the application's instructions directly on the hardware, not on top of the OS. This is called running on \textbf{bare metal}.
	\item The OS is \emph{not} executing while an application runs. It only gets the CPU back through an interrupt or a system call (Section~\ref{sec:interrupts}).
	\item So, for native (non-interpreted) code, whatever is currently running is running on bare metal.
\end{itemize}

\begin{figure}[H]
	\centering
	\begin{minipage}{0.4\textwidth}
		\centering
		\includegraphics[width=\textwidth]{images/img-01.png}
		\caption{The usual layered depiction}
	\end{minipage}
	\hfill
	\begin{minipage}{0.4\textwidth}
		\centering
		\includegraphics[width=\textwidth]{images/img-02.png}
		\caption{What is actually happening: the running code executes directly on the hardware}
	\end{minipage}
\end{figure}

\begin{extra}[Interpreters and virtual machines]
	Code run by an interpreter (e.g.\ Python, Java bytecode in a JVM) does not itself run on bare metal. The interpreter does, and it executes the code on its behalf. With hardware-assisted virtualization, a guest OS's instructions typically also run directly on the CPU, with a hypervisor stepping in only when needed.
\end{extra}

\subsection{Main Memory vs.\ Hard Disk}
\begin{itemize}
	\item Main memory is \textbf{byte addressable}.
	\item A hard disk is accessed in blocks called \textbf{sectors}. Positioning on a disk (seek and rotation) is slow compared to memory, so transferring one byte at a time would waste CPU time. Data is therefore moved in fixed-size chunks.
	\item Both are typically drawn as an array of slots for ease of understanding.
\end{itemize}

		 & \textbf{Main memory (RAM)} & \textbf{Hard disk} \\
		\midrule

		Relative speed & Fast & Slow \\
		Volatile? & Yes & No \\
		\bottomrule
	\end{tabular}
\end{table}

\section{x86 Privilege Rings}
\label{sec:rings}

\begin{figure}[H]
	\centering
	\begin{minipage}{0.3\textwidth}
		\centering
		\includegraphics[width=\textwidth]{images/img-05.png}
		\captionof{figure}{x86 protection rings}
	\end{minipage}
	\hfill
	\begin{minipage}{0.6\textwidth}
		\begin{itemize}
			\item The rings are privilege levels: \textbf{0 is the most privileged, 3 the least}.
			\item The CPU uses them to restrict which memory can be accessed and which instructions can be executed, so that buggy or malicious software cannot crash the system.
			\item \textbf{Ring 0 -- Kernel mode:} the OS kernel runs here with unrestricted access to memory and hardware.
			\item \textbf{Rings 1 \& 2:} intended for device drivers and OS services needing intermediate privilege.
			\item \textbf{Ring 3 -- User mode:} applications and user-space processes run here with restricted hardware access and isolated virtual memory.
		\end{itemize}
	\end{minipage}
\end{figure}

\begin{itemize}
	\item The current ring is kept by the CPU itself (the \emph{current privilege level}).
	\item Code cannot simply set itself to ring 0. The only ways to gain privilege are the controlled entry points: \textbf{interrupts, exceptions and system calls} (Section~\ref{sec:interrupts}). The CPU then jumps to OS code chosen by the OS, never to an address chosen by the application.
\end{itemize}

\begin{extra}[In practice]
	Mainstream operating systems (Windows, Linux) use only \textbf{ring 0 and ring 3}. Rings 1 and 2 exist in the architecture but are essentially unused.
\end{extra}

\section{Interrupts}
\label{sec:interrupts}

\begin{figure}[H]
	\centering
	\includegraphics[width=0.4\textwidth]{images/img-03.png}
	\caption{Devices connect to the CPU through interrupt lines}
\end{figure}

\subsection{Basics}
\begin{itemize}
	\item Interrupts are signals generated by I/O devices (keyboard, mouse, \dots) or by the timer. These devices are connected to the CPU through interrupt lines.
	\item The \reg{IVEC} (interrupt vector) register holds the memory address of the first instruction of the interrupt handler. The handler is part of the OS.
	\item After the handler finishes, execution must return to where it was interrupted. So the old \reg{PC} is saved on the stack before the handler starts, exactly like the return address of a function call (Section~\ref{sec:functions}).
\end{itemize}

\subsection{What Happens on an Interrupt}
\begin{enumerate}
	\item An application is running; the CPU is fetching its instructions.
	\item An interrupt signal arrives on an interrupt line. The CPU finishes the instruction it is executing.
	\item The old \reg{PC} (and flags) are saved on the stack.
	\item The ring level is set to 0, so the handler runs with the highest privilege.
	\item The value in \reg{IVEC} is copied into \reg{PC}.
	\item The next fetch reads the handler's first instruction (from the OS's memory) into \reg{IR}.
	\item The handler runs, so the OS is now running.
	\item At the end, a special return-from-interrupt instruction (\texttt{iret} on x86) restores the saved \reg{PC} and the previous ring, and the application resumes where it was interrupted.
\end{enumerate}
The exact order of the hardware steps differs between CPUs; what matters is that the old state is saved, privilege is raised, and \reg{PC} is redirected to the handler.

\begin{figure}[H]
	\centering
	\includegraphics[width=\textwidth]{images/img-04.png}
	\caption{Control transfer from application to OS on an interrupt (the return step is not shown)}
\end{figure}

\begin{pitfall}
	The interrupt changes \reg{PC}, not \reg{IR}. \reg{IR} merely holds the instruction currently being executed; it receives the handler's first instruction on the next fetch.
\end{pitfall}

\subsection{Why Interrupts Matter}
\begin{itemize}
	\item Without interrupts, an application could run forever and hog the CPU.
	\item Keyboard and mouse interrupts happen only when the user acts, so they cannot be relied on. The \textbf{timer} therefore raises an interrupt at regular intervals, guaranteeing that the OS regains control periodically.
\end{itemize}

\subsection{Asynchronous vs.\ Synchronous Interrupts}
\begin{table}[H]
	\centering
	\begin{tabular}{p{0.15\textwidth}p{0.38\textwidth}p{0.38\textwidth}}
		\toprule
		 & \textbf{Asynchronous} & \textbf{Synchronous} \\
		\midrule
		Caused by & External hardware (timer, keyboard, disk) & The running program's own instruction \\
		Timing & Can strike at any point; the program cannot predict it & Happens at a known instruction \\
		Program aware? & No, it does not necessarily know it happened & Yes \\
		Examples & Timer tick, key press, disk finished & System call (\texttt{int}/\texttt{syscall}), divide by zero, page fault \\
		Returns to & The instruction that was about to run & Usually the next instruction (system call); the same instruction for a fault that is fixed and retried \\
		\bottomrule
	\end{tabular}
\end{table}

\subsection{System Calls}
\label{sec:syscalls}
\begin{itemize}
	\item Privileged services (writing to the screen, disk or network) need ring~0, but applications run in ring~3. An application therefore asks the OS to do the work by raising a \textbf{synchronous interrupt} on purpose. This is a \textbf{system call}.
	\item On x86 this is done with the \texttt{int} instruction (e.g.\ \texttt{int 0x80} on 32-bit Linux) or the faster \texttt{syscall} instruction on x86-64. The idea is the same.
\end{itemize}

\begin{enumerate}
	\item The program calls a library function such as \texttt{write()}. This function is a \textbf{wrapper}: ordinary user-space code that does no real work itself.
	\item The wrapper puts the \textbf{system call number} (and the arguments) in registers and executes \texttt{int}/\texttt{syscall}.
	\item The CPU switches to ring~0 and jumps to the interrupt handler, as in Section~\ref{sec:interrupts}.
	\item The handler reads the system call number from the register and uses it to look up the matching kernel service (a table of function pointers).
	\item The kernel carries out the service on behalf of the process.
	\item The handler returns to user mode (\texttt{iret}/\texttt{sysret}), the wrapper returns, and the program continues.
\end{enumerate}

\begin{pitfall}[Wrapper vs.\ handler]
	The \emph{wrapper} (e.g.\ \texttt{write()} in the C library) is user code that triggers the interrupt. The privileged work is done by the kernel's implementation of the system call, to which the interrupt handler dispatches.
\end{pitfall}

\begin{extra}[Which stack does the handler use?]
	When an interrupt moves the CPU from ring~3 to ring~0, the CPU switches to a \textbf{kernel stack} (each process has one), because the user stack cannot be trusted: it may be too small, corrupted or deliberately malicious. A simplified model may say the handler uses the interrupted program's stack. When an interrupt occurs while the kernel is already running, no switch is needed.
\end{extra}

\begin{extra}[Real hardware]
	A single \reg{IVEC} register is a simplification. On x86 there are many interrupt sources, each with its own handler, so the CPU uses an \emph{interrupt vector table} (the IDT in protected mode), which holds one handler address per interrupt number.
\end{extra}

\section{Processes}
\label{sec:processes}

\subsection{Program vs.\ Process}
\begin{itemize}
	\item A \textbf{program} is a passive file on disk: instructions and data.
	\item A \textbf{process} is a program \emph{in execution}.
	\item A process is more than its instructions. To understand it fully you need its \textbf{state}: its code, its memory (data, heap, stack), the values of the registers including \reg{PC}, and the files it has open. Two processes can run the same program with different state.
	\item Each process has its own address space and its own stack in memory.
\end{itemize}

\subsection{Process Control Block (PCB)}
The OS keeps one \textbf{PCB} per process, a kernel data structure that records everything it needs to manage the process and to resume it later.

\begin{table}[H]
	\centering
	\begin{tabular}{ll}
		\toprule
		\textbf{PCB field} & \textbf{Purpose} \\
		\midrule
		Process ID (PID), parent PID & Identify the process and its parent \\
		State & Ready, running, blocked, \dots \\
		Saved \reg{PC} and registers & Restore the CPU exactly as the process left it \\
		Memory information & Where the process's memory is \\
		Open files & Files and devices the process is using \\
		\bottomrule
	\end{tabular}
\end{table}

\subsection{Parent and Child Processes}
\begin{itemize}
	\item New processes are created by an existing process (on Linux with \texttt{fork()}, often followed by \texttt{exec()}). The creator is the \textbf{parent}.
	\item Every process has a parent, except the very first one (\texttt{init}, PID 1), which the kernel starts itself (Section~\ref{sec:init}).
	\item If a parent terminates first, its children become \textbf{orphans} and are adopted by \texttt{init}.
\end{itemize}

\subsection{Process States and Context Switching}
\begin{itemize}
	\item The CPU never stops fetching and executing instructions while the machine is on, so there must always be something for it to run.
	\item A process that must wait for an event (e.g.\ a disk read) cannot continue, so it becomes \textbf{blocked}. The OS then runs another process that is \textbf{ready}.
	\item If no process is ready, the OS runs the \textbf{idle process}, which does nothing useful (conceptually \texttt{while(1)}; real kernels execute a halt instruction to save power) until an interrupt brings new work.
	\item The timer interrupt also limits how long a process may run. When its time is up, it goes back to the ready queue and another process gets the CPU.
\end{itemize}

	\centering
	\begin{tikzpicture}[>=Stealth,
		state/.style={draw, rounded corners, minimum width=2cm, minimum height=0.8cm, font=\small},

\paragraph{Context switch.} Moving the CPU from one process to another:
\begin{enumerate}
	\item An interrupt (timer or I/O) or a system call hands control to the OS.
	\item The OS saves the running process's \reg{PC} and registers into its PCB.
	\item The scheduler picks the next ready process.
	\item The OS loads that process's saved \reg{PC} and registers from its PCB and returns to it.
\end{enumerate}

\begin{pitfall}
	The OS does not ``ignore'' a waiting process or throw away its return address. All of its state is saved in the PCB, so the process can later continue exactly where it stopped.
\end{pitfall}

\section{Number Representation (Endianness)}

\begin{itemize}
	\item A number that spans multiple bytes has a \textbf{Most Significant Byte (MSB)} and a \textbf{Least Significant Byte (LSB)}.
	\item \emph{Endianness} defines the order in which these bytes are stored in memory.
	\item \textbf{Big endian:} the MSB is stored at the \emph{lowest} address.
	\item \textbf{Little endian:} the LSB is stored at the \emph{lowest} address.
	\item x86 uses \textbf{little endian}.
\end{itemize}

\subsection{Example}
Consider the 4-byte number \texttt{0x12345678}, with MSB $=$ \texttt{12} and LSB $=$ \texttt{78}.

\begin{table}[H]
	\centering
	\begin{tabular}{lcccc}
		\toprule
		\textbf{Address} & \texttt{+0} & \texttt{+1} & \texttt{+2} & \texttt{+3} \\
		\midrule
		Big endian & \texttt{12} & \texttt{34} & \texttt{56} & \texttt{78} \\
		Little endian & \texttt{78} & \texttt{56} & \texttt{34} & \texttt{12} \\
		\bottomrule
	\end{tabular}
\end{table}

\begin{extra}[Why it matters later]
	Endianness affects multi-byte fields on disk too. For example, the MBR signature (Section~\ref{sec:mbr}) is the byte sequence \texttt{55 AA}, which reads as the 16-bit value \texttt{0xAA55} on a little-endian machine.
\end{extra}

\section{Function Calls and the Stack}
\label{sec:functions}

\subsection{Saving Registers}
\begin{itemize}
	\item When a function is going to use a register that already holds a value the caller needs, the compiler emits instructions that \texttt{push} the register onto the stack at the start of the function and \texttt{pop} it back at the end.
	\item This preserves the caller's values.
	\item The exact rules (whether caller or callee saves which registers) are defined by the \textbf{calling convention}.
\end{itemize}

\begin{pitfall}
	Declaring a variable does not by itself push a register. Local variables are either kept in registers or allocated in the function's stack frame (by adjusting the stack pointer). Pushes happen to \emph{save registers that the function will overwrite}.
\end{pitfall}

\subsection{Example: One-Register Machine}
\begin{minted}{c}
#include <stdio.h>

void func() {
	int x = 0;
	x++;
	printf("%d\n", x);
}

int main() {
	int b = 10;
	func();
	printf("%d\n", b);
	return 0;
}
\end{minted}
The output is \texttt{1} followed by \texttt{10}.

Assume the machine has only one register, \reg{R}, for holding variable values.

\begin{enumerate}
	\item \texttt{main} runs first and stores \texttt{b = 10} in \reg{R}.
	\item The call instruction pushes the \textbf{return address} onto the stack, then sets \reg{PC} to the address of the first instruction of \texttt{func}.
	\item At the start of \texttt{func}, before \reg{R} is reused for \texttt{x}, the old value of \reg{R} (10) is pushed onto the stack.
	\item \texttt{func} sets \reg{R} to 0, increments it to 1 and prints it.
	\item Before returning, \texttt{func} pops the saved value back into \reg{R}, so \reg{R} holds 10 again.
	\item The return instruction pops the return address into \reg{PC}, and \texttt{main} resumes where it left off and prints 10.
\end{enumerate}

\begin{table}[H]
	\centering
	\begin{tabular}{cl}
		\toprule
		\textbf{Stack (top first)} & \textbf{State} \\
		\midrule
		\texttt{saved R = 10} & after the push inside \texttt{func} \\
		\texttt{return address} & pushed by the call \\
		\bottomrule
	\end{tabular}
\end{table}
Values are popped in the reverse order they were pushed (LIFO): the saved register first, then the return address.

\subsection{Arguments and Return Values}
The calling convention also says how arguments and the result are passed. On \textbf{32-bit x86} (the \texttt{cdecl} convention) the caller pushes the arguments on the stack, and the result comes back in a register (\texttt{eax}). On x86-64 the first few arguments are passed in registers instead, and the stack is used for extra ones.

\begin{minted}{c}
#include <stdio.h>

int add(int a, int b) {
	return a + b;
}

int main() {
	int r = add(2, 3);
	printf("%d\n", r);
	return 0;
}
\end{minted}

\begin{enumerate}
	\item The caller pushes the arguments, last one first: \texttt{3}, then \texttt{2}.
	\item The call instruction pushes the return address and sets \reg{PC} to \texttt{add}.
	\item \texttt{add} reads \texttt{a} and \texttt{b} from fixed positions relative to the stack pointer, and leaves the sum in \texttt{eax}.
	\item The return instruction pops the return address into \reg{PC}.
	\item The caller removes the two arguments from the stack and copies \texttt{eax} into \texttt{r}.
\end{enumerate}

\begin{table}[H]
	\centering
	\begin{tabular}{cl}
		\toprule
		\textbf{Stack (top first)} & \textbf{Pushed by} \\
		\midrule
		\texttt{return address} & the call instruction \\
		\texttt{a = 2} & the caller \\
		\texttt{b = 3} & the caller \\
		\bottomrule
	\end{tabular}
\end{table}

\section{Booting}

\begin{figure}[H]
	\centering

\subsection{Power-On}
\begin{itemize}
	\item RAM contents are undefined at power-on. Most registers also hold arbitrary values, but the hardware resets a few (including \reg{PC}) to defined values.
	\item \reg{PC} is set to a fixed \textbf{reset vector} in ROM. ROM is non-volatile, so unlike RAM its contents survive power loss.
	\item On classic x86 the reset vector is 16 bytes below the top of the address space. These 16 bytes hold a jump to the entry code of the \textbf{BIOS} firmware. This indirection allows different machines to have different ROM sizes and layouts.
	\item The BIOS runs in real mode with unrestricted hardware access (the privilege rings do not apply yet).
\end{itemize}

\subsection{Boot Sequence}
\begin{enumerate}
	\item \textbf{POST} (Power-On Self Test): the BIOS checks that the hardware is present and working.
	\item The BIOS reads the \textbf{first sector} of the boot disk into RAM. This is the MBR (Section~\ref{sec:mbr}).
	\item The BIOS jumps to the start of that code in RAM (the \emph{boot call}).
	\item The MBR code loads a \textbf{bootloader} (directly, or via a second stage or the active partition's boot sector).
	\item The bootloader loads the \textbf{OS kernel} into memory and jumps to it.
	\item The kernel initializes the system (Section~\ref{sec:kernel-boot}).
	\item The kernel starts the first user process, \texttt{init} (Section~\ref{sec:init}).
\end{enumerate}

\begin{extra}[UEFI]
	Modern machines use UEFI firmware instead of the legacy BIOS and typically a GPT partition scheme instead of the MBR. The overall idea (firmware $\rightarrow$ bootloader $\rightarrow$ kernel) is the same.
\end{extra}

\subsection{Master Boot Record (MBR)}
\label{sec:mbr}
Consider a disk with 512-byte sectors. The MBR is the first sector.

\begin{table}[H]
	\centering
	\begin{tabular}{lll}
		\toprule
		\textbf{Offset} & \textbf{Size} & \textbf{Content} \\
		\midrule
		0 & 446 bytes & Boot code \\
		446 & 64 bytes & Partition table (4 entries $\times$ 16 bytes) \\
		510 & 2 bytes & Boot signature: \texttt{55} at 510, \texttt{AA} at 511 (written 0x55AA) \\
		\bottomrule
	\end{tabular}
\end{table}

\begin{itemize}
	\item The signature marks the sector as bootable. Read as a little-endian 16-bit word it is \texttt{0xAA55}.
	\item The table holds up to \textbf{4 primary partitions}.
\end{itemize}

\paragraph{Partition entry (16 bytes).}
\begin{table}[H]
	\centering
	\begin{tabular}{lll}
		\toprule
		\textbf{Field} & \textbf{Size} & \textbf{Meaning} \\
		\midrule
		Status & 1 B & \texttt{0x80} = active (bootable), \texttt{0x00} = inactive \\
		Start CHS & 3 B & Start address in cylinder/head/sector form \\
		Type & 1 B & Partition type code \\
		End CHS & 3 B & End address in CHS form \\
		Start LBA & 4 B & First sector, as a logical block address \\
		Sector count & 4 B & Size of the partition in sectors \\
		\bottomrule
	\end{tabular}
\end{table}
\begin{itemize}
	\item There is no stored partition number; it is implied by the entry's position in the table.
	\item The type is a \emph{code}, not a filesystem name. For example \texttt{0x07} is NTFS/exFAT and \texttt{0x83} is Linux (used for ext4). A code can map to several filesystems.
\end{itemize}

\begin{extra}[Magic numbers]
	A \textbf{magic number} is a fixed byte pattern at the start of a file that identifies its format. On Linux, the kernel uses it to decide how to run an executable, and the \texttt{file} command uses it (plus other heuristics) to identify files. The file extension does not matter.
	\begin{itemize}
		\item ELF executables: \texttt{0x7F} followed by \texttt{ELF}
		\item Java class files: \texttt{0xCAFEBABE}
		\item Scripts: \texttt{\#!} (the magic) followed by an interpreter path, e.g.\ \texttt{\#!/bin/bash}. The path can be any interpreter.
	\end{itemize}
\end{extra}

\subsection{Kernel Initialization}
\label{sec:kernel-boot}
Once the bootloader hands over control, the kernel (running in ring~0) prepares the machine for use:
\begin{itemize}
	\item It sets up its own interrupt handling: it fills in \reg{IVEC} (the interrupt vector table) and configures the \textbf{timer} so that it regains control periodically.
	\item It initializes memory management, the scheduler and the device drivers.
	\item It \textbf{mounts} the root file system, so that files can be read.
	\item Finally it starts the first user process, \texttt{init}.
\end{itemize}

\begin{pitfall}
	Creating file systems (formatting) and configuring the network are not part of the kernel's boot work. Mounting the root file system is. Network and most service configuration is done later by user-space programs started by \texttt{init}/\texttt{systemd}.
\end{pitfall}

\subsection{Init / Systemd}
\label{sec:init}
\begin{itemize}
	\item After the kernel has finished initializing, the first \emph{user-space} process it starts is \texttt{init} (or \texttt{systemd} on newer Linux systems).
	\item The CPU is now switched to \textbf{ring~3}, the lowest privilege level, for this process.
	\item Since it is the first process, it has \textbf{PID 1}.
	\item It is the only user-space process the kernel starts directly. \texttt{init} then starts all the others: the classic SysV \texttt{init} reads \texttt{/etc/inittab}, while \texttt{systemd} reads its unit files (e.g.\ under \texttt{/etc/systemd/system}).
	\item It is also the adoptive parent of orphaned processes (Section~\ref{sec:processes}).
\end{itemize}

\begin{pitfall}[Root is not ring 0]
	Processes started by \texttt{init} may run as the \textbf{root} user (superuser), but they still execute in \textbf{ring~3}. Root is an \emph{OS-level permission} (what the kernel allows), whereas the ring is a \emph{CPU-level privilege} (what the hardware allows). A root process still needs system calls to reach the hardware.
\end{pitfall}

\begin{extra}[Kernel threads]
	The kernel also creates its own threads that run entirely in kernel mode, e.g.\ the idle task (PID 0) and \texttt{kthreadd} (PID 2). They are not user-space processes, which is why \texttt{init} is the only \emph{user} process started by the kernel.
\end{extra}
